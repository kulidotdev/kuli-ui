### Success View

```tsx
import { SignUp, SignupSuccessView } from "@/components/auth/signup";

export default function SignUpPage() {
  const [success, setSuccess] = useState(false);

  if (success) return <SignupSuccessView redirectUrl="/signin" />;

  return (
    <SignUp onSubmit={async (values) => { /* ... */ setSuccess(true); }}>
      ...
    </SignUp>
  );
}
```

## Types

```ts
type SignupFormValues = {
  name: string;
  email: string;
  password: string;
  username?: string;  // only present when showUsername={true}
  phone?: string;     // E.164 format, only present when showPhone={true}
};
```

## Validation

| Field | Rule |
|---|---|
| `name` | `min(1)` |
| `email` | Valid email format |
| `password` | `min(8)`, must contain uppercase, lowercase, and a digit |
| `username` | `min(3)`, `max(20)`, alphanumeric + underscore |
| `phone` | E.164 via `isValidPhoneNumber()` |

<Callout type="info">
  `username` and `phone` are **optional** in the Zod schema — only validated if the user fills them in.
</Callout>
