## Usage

### Email Only (Default)

```tsx
import { ForgotPassword, ForgotPasswordSuccessView } from "@/components/auth/forgot-password";
import type { ForgotPasswordEmailValues } from "@/components/auth/forgot-password-types";

export default function ForgotPasswordPage() {
  const [isLoading, setIsLoading] = useState(false);
  const [sent, setSent] = useState(false);
  const [apiError, setApiError] = useState<{ message: string } | null>(null);

  const handleEmail = async (values: ForgotPasswordEmailValues) => {
    setIsLoading(true);
    setApiError(null);
    try {
      await yourAuthLibrary.sendPasswordResetEmail(values.email);
      setSent(true);
    } catch (err) {
      setApiError({ message: err.message ?? "Failed to send." });
    } finally {
      setIsLoading(false);
    }
  };

  if (sent) return <ForgotPasswordSuccessView backUrl="/signin" />;

  return (
    <ForgotPassword onSubmitEmail={handleEmail} isLoading={isLoading} apiError={apiError}>
      <ForgotPassword.Header />
      <ForgotPassword.Content>
        <ForgotPassword.Tabs>
          <ForgotPassword.EmailForm>
            <ForgotPassword.EmailField />
            <ForgotPassword.SubmitButton />
          </ForgotPassword.EmailForm>
        </ForgotPassword.Tabs>
      </ForgotPassword.Content>
    </ForgotPassword>
  );
}
```

### With Phone Support (OTP)

```tsx
import {
  ForgotPassword,
  ForgotPasswordSuccessView,
} from "@/components/auth/forgot-password";
import type {
  ForgotPasswordEmailValues,
  ForgotPasswordPhoneValues,
} from "@/components/auth/forgot-password-types";

export default function ForgotPasswordPage() {
  const [isLoading, setIsLoading] = useState(false);
  const [sent, setSent] = useState(false);
  const [apiError, setApiError] = useState<{ message: string } | null>(null);

  const handleEmail = async (values: ForgotPasswordEmailValues) => {
    setIsLoading(true);
    setApiError(null);
    try {
      await yourAuthLibrary.sendPasswordResetEmail(values.email);
      setSent(true);
    } catch (err) {
      setApiError({ message: err.message ?? "Failed to send reset link." });
    } finally {
      setIsLoading(false);
    }
  };

  const handlePhone = async (values: ForgotPasswordPhoneValues) => {
    setIsLoading(true);
    setApiError(null);
    try {
      await yourAuthLibrary.sendOtp(values.phone);
      setSent(true);
    } catch (err) {
      setApiError({ message: err.message ?? "Failed to send OTP." });
    } finally {
      setIsLoading(false);
    }
  };

  if (sent) return <ForgotPasswordSuccessView backUrl="/signin" />;

  return (
    <ForgotPassword
      onSubmitEmail={handleEmail}
      onSubmitPhone={handlePhone}
      isLoading={isLoading}
      apiError={apiError}
      allowPhone
    >
      <ForgotPassword.Header />
      <ForgotPassword.Content>
        <ForgotPassword.Tabs>
          <ForgotPassword.EmailForm>
            <ForgotPassword.EmailField />
            <ForgotPassword.SubmitButton />
          </ForgotPassword.EmailForm>
          <ForgotPassword.PhoneForm>
            <ForgotPassword.PhoneField />
            <ForgotPassword.SubmitButton />
          </ForgotPassword.PhoneForm>
        </ForgotPassword.Tabs>
      </ForgotPassword.Content>
    </ForgotPassword>
  );
}
```

## Props

### `<ForgotPassword>` (Root Provider)

| Prop | Type | Default | Description |
|---|---|---|---|
| `onSubmitEmail` | `(values: ForgotPasswordEmailValues) => Promise<void>` | — | **Required.** Email submit callback |
| `onSubmitPhone` | `(values: ForgotPasswordPhoneValues) => Promise<void>` | — | Phone submit callback. Required when `allowPhone={true}` |
| `isLoading` | `boolean` | `false` | Disables all inputs |
| `apiError` | `{ message: string; code?: string } \| null` | `null` | Server error to display |
| `allowPhone` | `boolean` | `false` | Enable the phone tab with a PhoneInput |
| `children` | `ReactNode` | — | **Required.** |

### `<ForgotPassword.Header>`

| Prop | Type | Default | Description |
|---|---|---|---|
| `title` | `ReactNode` | `"Forgot Password"` | Card title |
| `description` | `ReactNode` | Auto (varies per tab) | Subtitle. Changes based on `activeTab` and `allowPhone` |

### `<ForgotPassword.SubmitButton>`

| Prop | Type | Default | Description |
|---|---|---|---|
| `children` | `ReactNode` | Auto | `"Send Reset Link"` on email tab, `"Send OTP"` on phone tab |

### `<ForgotPasswordSuccessView>`

| Prop | Type | Description |
|---|---|---|
| `backUrl` | `string` | **Required.** URL for the "Go Back" button |

## Types

```ts
type ForgotPasswordEmailValues = {
  email: string;
};

type ForgotPasswordPhoneValues = {
  phone: string; // E.164 format
};
```

## Behaviour

- When `allowPhone={false}` (default), `ForgotPassword.Tabs` renders children directly with no tab UI.
- When `allowPhone={true}`, a `<Tabs>` with "Email" and "Phone" triggers is shown. The inactive form returns `null` automatically.
- The submit button adapts its label based on the `activeTab`.
- `ForgotPasswordSuccessView` is standalone — manage the `sent` state outside and swap to it yourself.

## Sub-Components

| Sub-component | Description |
|---|---|
| `<ForgotPassword.Header>` | Card header |
| `<ForgotPassword.Content>` | Card content, automatically renders `apiError` |
| `<ForgotPassword.Tabs>` | Tab wrapper — renders children directly when `allowPhone={false}` |
| `<ForgotPassword.EmailForm>` | `<form>` wrapper for the email flow |
| `<ForgotPassword.PhoneForm>` | `<form>` wrapper for the phone flow |
| `<ForgotPassword.EmailField>` | Email input |
| `<ForgotPassword.PhoneField>` | Phone input with country picker |
| `<ForgotPassword.SubmitButton>` | Adaptive submit button |
| `<ForgotPasswordSuccessView>` | Standalone success screen |
