## Usage

### Email Flow (Reset via URL Token)

```tsx
import { ResetPassword, ResetPasswordSuccessView } from "@/components/auth/reset-password";
import type { ResetPasswordEmailValues } from "@/components/auth/forgot-password-types";

export default function ResetPasswordPage() {
  const searchParams = useSearchParams();
  const token = searchParams.get("token") ?? "";
  const [isLoading, setIsLoading] = useState(false);
  const [done, setDone] = useState(false);
  const [apiError, setApiError] = useState<{ message: string } | null>(null);

  const handleSubmit = async (values: ResetPasswordEmailValues) => {
    setIsLoading(true);
    setApiError(null);
    try {
      await yourAuthLibrary.resetPassword({ token, newPassword: values.password });
      setDone(true);
    } catch (err) {
      setApiError({ message: err.message ?? "Reset failed." });
    } finally {
      setIsLoading(false);
    }
  };

  if (done) return <ResetPasswordSuccessView redirectTo="/signin" />;

  return (
    <ResetPassword method="email" onSubmitEmail={handleSubmit} isLoading={isLoading} apiError={apiError}>
      <ResetPassword.Header />
      <ResetPassword.Content>
        <ResetPassword.Form>
          <ResetPassword.PasswordField />
          <ResetPassword.ConfirmPasswordField />
          <ResetPassword.SubmitButton />
        </ResetPassword.Form>
      </ResetPassword.Content>
    </ResetPassword>
  );
}
```

### Phone Flow (Reset via OTP)

```tsx
import { ResetPassword, ResetPasswordSuccessView } from "@/components/auth/reset-password";
import type { ResetPasswordPhoneValues } from "@/components/auth/forgot-password-types";

export default function ResetPasswordPhonePage() {
  const [isLoading, setIsLoading] = useState(false);
  const [done, setDone] = useState(false);
  const [apiError, setApiError] = useState<{ message: string } | null>(null);

  const handleSubmit = async (values: ResetPasswordPhoneValues) => {
    setIsLoading(true);
    setApiError(null);
    try {
      await yourAuthLibrary.resetPasswordWithOtp({ otp: values.code, newPassword: values.password });
      setDone(true);
    } catch (err) {
      setApiError({ message: err.message ?? "Reset failed." });
    } finally {
      setIsLoading(false);
    }
  };

  if (done) return <ResetPasswordSuccessView redirectTo="/signin" />;

  return (
    <ResetPassword
      method="phone"
      onSubmitPhone={handleSubmit}
      isLoading={isLoading}
      apiError={apiError}
    >
      <ResetPassword.Header description="Enter the OTP sent to your phone and choose a new password." />
      <ResetPassword.Content>
        <ResetPassword.Form>
          <ResetPassword.OtpField />
          <ResetPassword.PasswordField />
          <ResetPassword.ConfirmPasswordField />
          <ResetPassword.SubmitButton />
        </ResetPassword.Form>
      </ResetPassword.Content>
    </ResetPassword>
  );
}
```

## Props

### `<ResetPassword>` (Root Provider)

| Prop | Type | Default | Description |
|---|---|---|---|
| `method` | `"email" \| "phone"` | `"email"` | Determines the flow |
| `onSubmitEmail` | `(values: ResetPasswordEmailValues) => Promise<void>` | — | Required when `method="email"` |
| `onSubmitPhone` | `(values: ResetPasswordPhoneValues) => Promise<void>` | — | Required when `method="phone"` |
| `isLoading` | `boolean` | `false` | Disables all inputs |
| `apiError` | `{ message: string; code?: string } \| null` | `null` | Server error to display |
| `children` | `ReactNode` | — | **Required.** |

### `<ResetPassword.Header>`

| Prop | Type | Default | Description |
|---|---|---|---|
| `title` | `ReactNode` | `"Reset Password"` | Card title |
| `description` | `ReactNode` | `"Enter your new password below."` | Card subtitle |

### `<ResetPassword.OtpField>`

Only rendered when `method="phone"`. Displays a 6-slot `InputOTP`.

### `<ResetPasswordSuccessView>`

| Prop | Type | Description |
|---|---|---|
| `redirectTo` | `string` | **Required.** URL for the "Continue" button |

## Types

```ts
type ResetPasswordEmailValues = {
  password: string;
  confirmPassword: string;
};

type ResetPasswordPhoneValues = {
  code: string;            // 6-digit OTP
  password: string;
  confirmPassword: string;
};
```

## Validation

| Field | Rule |
|---|---|
| `code` | Exactly 6 digits (phone flow only) |
| `password` | `min(8)` |
| `confirmPassword` | Must match `password` via `z.refine()` |

## Sub-Components

| Sub-component | Visible when | Description |
|---|---|---|
| `<ResetPassword.Header>` | Always | Card header |
| `<ResetPassword.Content>` | Always | Card content, automatically renders `apiError` |
| `<ResetPassword.Form>` | Always | `<form>` wired to the correct flow based on `method` |
| `<ResetPassword.OtpField>` | `method="phone"` | 6-slot OTP input |
| `<ResetPassword.PasswordField>` | Always | New password input |
| `<ResetPassword.ConfirmPasswordField>` | Always | Confirm password input |
| `<ResetPassword.SubmitButton>` | Always | Submit with loading state |
| `<ResetPasswordSuccessView>` | — | Standalone success screen |
